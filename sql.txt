CREATE TABLE `users` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `user_id` varchar(255) NOT NULL,
  `name` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `role_id` int(11) NOT NULL,
  `status` enum('Active','Inactive','Suspended') NOT NULL,
  `last_login` datetime NOT NULL DEFAULT current_timestamp(),
  `created_by` varchar(255) NOT NULL,
  `created_at` datetime NOT NULL DEFAULT current_timestamp(),
  `updated_at` datetime NOT NULL DEFAULT current_timestamp(),
  PRIMARY KEY (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci

CREATE TABLE `nominees` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `emp_id` int(11) NOT NULL,
  `nominee_name` varchar(255) NOT NULL,
  `age` int(11) DEFAULT NULL,
  `relationship` varchar(50) NOT NULL,
  `nominee_address` text DEFAULT NULL,
  `percentage` decimal(5,2) DEFAULT 100.00,
  `contact_no` varchar(20) DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `fk_nominee_emp` (`emp_id`),
  CONSTRAINT `fk_nominee_emp` FOREIGN KEY (`emp_id`) REFERENCES `employee_details` (`emp_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci

CREATE TABLE `attendance` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `emp_id` int(11) NOT NULL,
  `date` date NOT NULL,
  `status` enum('Present','Half Day','Leave','Absent') DEFAULT 'Present',
  `check_in_time` time DEFAULT NULL,
  `check_out_time` time DEFAULT NULL,
  `notes` text DEFAULT NULL,
  `created_at` datetime DEFAULT current_timestamp(),
  `updated_at` datetime DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  PRIMARY KEY (`id`),
  UNIQUE KEY `emp_date_idx` (`emp_id`,`date`),
  KEY `fk_attendance_emp` (`emp_id`),
  CONSTRAINT `fk_attendance_emp` FOREIGN KEY (`emp_id`) REFERENCES `employee_details` (`emp_id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;



CREATE TABLE IF NOT EXISTS `crane_logbook` (
  `id` INT AUTO_INCREMENT PRIMARY KEY,
  
  -- Date & Period
  `month` VARCHAR(30) NOT NULL COMMENT 'e.g. August 2026 or 2026-08',
  `logbook_date` DATE NOT NULL COMMENT 'Date of the log entry',
  
  -- References / IDs
  `site_id` INT NOT NULL COMMENT 'Reference to sites table',
  `crane_number_id` INT NOT NULL COMMENT 'Reference to cranes table (crane_id)',
  
  -- Time Duration
  `start_time` VARCHAR(30) NULL COMMENT 'Work start time, e.g. 08:00 AM or 08:00',
  `end_time` VARCHAR(30) NULL COMMENT 'Work end time, e.g. 05:00 PM or 17:00',
  `total_hours` DECIMAL(6, 2) GENERATED ALWAYS AS (
    CASE 
      WHEN `hours_end` >= `hours_start` THEN (`hours_end` - `hours_start`) 
      ELSE NULL 
    END
  ) VIRTUAL COMMENT 'Calculated working hours',
  
  -- Hour Meter Readings (HMR)
  `hours_start` DECIMAL(10, 2) NULL COMMENT 'HMR Start Reading',
  `hours_end` DECIMAL(10, 2) NULL COMMENT 'HMR End Reading',
  
  -- KM / KMH Readings (Odometer)
  `kmh_start` DECIMAL(10, 2) NULL COMMENT 'KM / Speedometer Start Reading',
  `kmh_end` DECIMAL(10, 2) NULL COMMENT 'KM / Speedometer End Reading',
  `total_kmh` DECIMAL(10, 2) GENERATED ALWAYS AS (
    CASE 
      WHEN `kmh_end` >= `kmh_start` THEN (`kmh_end` - `kmh_start`) 
      ELSE NULL 
    END
  ) VIRTUAL COMMENT 'Calculated total distance/run',
  
  -- Work & Operational Details
  `work_description` TEXT NOT NULL COMMENT 'Description of work carried out',
  
  -- Signatures & Approvals
  `site_incharge_sign` VARCHAR(255) NULL COMMENT 'Incharge signature (name, status, or file URL)',
  `operator_sign` VARCHAR(255) NULL COMMENT 'Operator signature (name, status, or file URL)',
  
  -- Remarks & System Metadata
  `remarks` TEXT NULL COMMENT 'Any additional observations or breakdown notes',
  `created_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
  `updated_at` TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,

  -- Indexes for high-speed queries & filtering
  INDEX `idx_logbook_date` (`logbook_date`),
  INDEX `idx_month` (`month`),
  INDEX `idx_site_id` (`site_id`),
  INDEX `idx_crane_number_id` (`crane_number_id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
